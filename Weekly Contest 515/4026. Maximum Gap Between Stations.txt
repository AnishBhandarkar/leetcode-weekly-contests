4026. Maximum Gap Between Stations


/**
 * @param {string} skill
 * @param {string} station
 * @return {number}
 */
var maximumGap = function (skill, station) {
    const n = skill.length;

    const left = new Array(n);
    const right = new Array(n);

    let j = 0;
    for (let i = 0; i < n; i++) {
        while (station[j] !== skill[i]) {
            j++;
        }
        left[i] = j;
        j++;
    }

    j = station.length - 1;
    for (let i = n - 1; i >= 0; i--) {
        while (station[j] !== skill[i]) {
            j--;
        }
        right[i] = j;
        j--;
    }

    let result = 0;
    for (let i = 0; i < n - 1; i++) {
        result = Math.max(result, right[i + 1] - left[i]);
    }

    return result;
};

/*
T(n) = O(n + m)
S(n) = O(n)
*/
